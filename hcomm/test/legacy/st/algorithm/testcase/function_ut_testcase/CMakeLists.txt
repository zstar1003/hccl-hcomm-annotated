add_executable(legacy_alg_function_ut_testcase ${test_case_list})
 
#头文件搜索路径
target_include_directories(legacy_alg_function_ut_testcase PRIVATE
    ${include_list}
    ${UTILS_DIR}/checker/task_collector
    ${UTILS_DIR}/adapter_v2
    ${UTILS_DIR}/checker/mem_conflict_check
    ${HCOMM_DIR}/src/legacy/ascend910/common
    ${HCOMM_DIR}/src/legacy/ascend910/common/error_manager
)
 
target_compile_definitions(legacy_alg_function_ut_testcase PRIVATE
    google=ascend_private
    -DOPEN_HCCL_TEST
    _GLIBCXX_USE_CXX11_ABI=0
    -DHCCL_ALG_ANALYZER_DAVID
)
 
target_compile_options(legacy_alg_function_ut_testcase PRIVATE
    -U_FORTIFY_SOURCE
    -Os
    -O0 -g
    $<$<STREQUAL:${ENABLE_ASAN},true>:-fsanitize=address -fsanitize-recover=address,all -fno-omit-frame-pointer>
)

add_dependencies(legacy_alg_function_ut_testcase
    hccl_alg_adaption_v2
)
 
target_link_libraries(legacy_alg_function_ut_testcase PRIVATE
    $<BUILD_INTERFACE:intf_llt_pub>
    $<BUILD_INTERFACE:rdma_core_headers>
    hccl_alg_analyzer_legacy
    hccl_orion_alg_test
    hccl_alg_adaption_v2
    ${ASCEND_CANN_PACKAGE_PATH}/lib64/libc_sec.so
    $<$<STREQUAL:${ENABLE_ASAN},true>:-fsanitize=address>
    -pthread
    ascend_protobuf
    -lrt
    -ldl
)

target_sources(legacy_alg_function_ut_testcase
    PUBLIC
    main.cc
    ${UTILS_DIR}/common/virtual_topo_stub/virtual_topo_stub.cc
    testcase_utils.cc
    gen_ccu_task_node_utils.cc
    testcase_ccu_revamp_parallel.cc
    testcase_ccu_revamp_bilateral.cc
    testcase_selector.cc
)

enable_testing()
add_test(NAME legacy_alg_function_ut_testcase
         COMMAND legacy_alg_function_ut_testcase
         WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
)
